    28→export function PlansView() {
    29→  const { toast } = useToast()
    30→  const setCaptureOpen = useUI((s) => s.setCaptureOpen)
    31→  const { data, loading, reload } = useApi<{ plans: PlanNode[] }>('/api/plans')
    32→  const [mode, setMode] = useState<'outline' | 'kanban' | 'day'>('outline')
    33→  const [selectedDay, setSelectedDay] = useState(todayISO())
    34→  const [addOpen, setAddOpen] = useState(false)
    35→  const [templatesOpen, setTemplatesOpen] = useState(false)
    36→  const [collapsed, setCollapsed] = useState<Set<string>>(new Set())
    37→
    38→  const week = useMemo(() => {
    39→    const now = new Date()
    40→    const day = (now.getDay() + 6) % 7 // Monday=0
    41→    const monday = new Date(now)
    42→    monday.setDate(now.getDate() - day)
    43→    return Array.from({ length: 7 }).map((_, i) => {
    44→      const d = new Date(monday)
    45→      d.setDate(monday.getDate() + i)
    46→      return d
    47→    })
    48→  }, [])
    49→
    50→  const dayTasks = useApi<{ tasks: Task[] }>(`/api/tasks?date=${selectedDay}`, [selectedDay])
    51→
    52→  async function toggleTask(t: Task) {
    53→    const next = t.status === 'done' ? 'todo' : 'done'
    54→    try {
    55→      await api.patch(`/api/tasks/${t.id}`, { status: next })
    56→      reload()
    57→      dayTasks.reload()
    58→    } catch {
    59→      toast({ title: 'Could not update task', variant: 'destructive' })
    60→    }
    61→  }
    62→
    63→  async function snoozeTask(t: Task) {
    64→    try {
    65→      await api.patch(`/api/tasks/${t.id}`, { snooze: 1, baseDue: t.dueDate ?? todayISO() })
    66→      toast({ title: 'Snoozed to tomorrow' })
    67→      reload()
    68→      dayTasks.reload()
    69→    } catch {
    70→      toast({ title: 'Could not snooze', variant: 'destructive' })
    71→    }
    72→  }
    73→
    74→  async function addTaskToPlan(planId: string | null, title: string) {
    75→    try {
    76→      await api.post('/api/tasks', { title, planId, dueDate: planId ? undefined : selectedDay })
    77→      reload()
    78→      dayTasks.reload()
    79→    } catch {
    80→      toast({ title: 'Could not add task', variant: 'destructive' })
    81→    }
    82→  }
    83→
    84→  function toggleCollapse(id: string) {
    85→    setCollapsed((prev) => {
    86→      const next = new Set(prev)
    87→      if (next.has(id)) next.delete(id)
    88→      else next.add(id)
    89→      return next
    90→    })
    91→  }
    92→
    93→  const allPlansFlat = useMemo(() => {
    94→    const out: { id: string; title: string; timeframe: string }[] = []
    95→    const walk = (nodes: PlanNode[]) => {
    96→      for (const n of nodes) {
    97→        out.push({ id: n.id, title: `${n.timeframe === 'day' ? '☀' : n.timeframe === 'week' ? '🗓' : n.timeframe === 'month' ? '📅' : '🎯'} ${n.title}`, timeframe: n.timeframe })
    98→        walk((n.children as PlanNode[]) ?? [])
    99→      }
   100→    }
   101→    walk(data?.plans ?? [])
   102→    return out
   103→  }, [data])
   104→
   105→  if (loading || !data) {
   106→    return (
   107→      <div className="space-y-4 pb-8">
   108→        <SkeletonCard className="h-20" />
   109→        <SkeletonCard className="h-64" />
   110→      </div>
   111→    )
   112→  }
   113→
   114→  return (
   115→    <div className="anim-fade-up space-y-4 pb-8">
   116→      {/* Header */}
   117→      <div className="flex flex-wrap items-center justify-between gap-3">
   118→        <div>
   119→          <h1 className="text-2xl font-bold tracking-tight">Plans</h1>
   120→          <p className="text-sm text-muted-foreground">Year → Month → Week → Day, with milestones and tasks.</p>
   121→        </div>
   122→        <div className="flex items-center gap-2">
   123→          <div className="flex overflow-hidden rounded-lg border">
   124→            <button onClick={() => setMode('outline')} className={cn('flex h-9 items-center gap-1.5 whitespace-nowrap px-3 text-xs font-medium transition-colors', mode === 'outline' ? 'bg-primary/10 text-primary' : 'text-muted-foreground hover:bg-muted')} aria-label="Outline view">
   125→              <LayoutList className="h-4 w-4" /> Outline
   126→            </button>

   128→              <KanbanSquare className="h-4 w-4" /> Kanban
   129→            </button>
   130→          </div>
   131→          <Button variant="outline" onClick={() => setTemplatesOpen(true)}>
   132→            <Wand2 className="mr-1.5 h-4 w-4" /> Templates
   133→          </Button>
   134→          <Button onClick={() => setAddOpen(true)}>
   135→            <Plus className="mr-1.5 h-4 w-4" /> Plan
   136→          </Button>
   137→        </div>
   138→      </div>
   139→
   140→      {/* Week strip — drag tasks here to reschedule */}
   141→      <div className="grid grid-cols-7 gap-1.5 sm:gap-2">
   142→        {week.map((d) => {
   143→          const iso = d.toISOString().slice(0, 10)
   144→          const isToday = iso === todayISO()
   145→          const isSel = iso === selectedDay
   146→          return (
   147→            <button
   148→              key={iso}
   149→              onClick={() => { setSelectedDay(iso); setMode('day') }}
   150→              onDragOver={(e) => e.preventDefault()}
   151→              onDrop={(e) => {
   152→                e.preventDefault()
   153→                const taskId = e.dataTransfer.getData('text/task-id')

   155→                  api.patch(`/api/tasks/${taskId}`, { dueDate: `${iso}T09:00:00` })

   157→                    .catch(() => toast({ title: 'Reschedule failed', variant: 'destructive' }))
   158→                }
   159→              }}
   160→              className={cn(
   161→                'flex min-h-[64px] flex-col items-center justify-center gap-0.5 rounded-xl border bg-card p-1.5 transition-colors sm:min-h-[72px]',
   162→                isSel && 'border-primary bg-primary/5',
   163→                isToday && !isSel && 'border-primary/40'
   164→              )}
   165→            >
   166→              <span className="text-[10px] font-medium uppercase text-muted-foreground">{d.toLocaleDateString('en-US', { weekday: 'short' })}</span>
   167→              <span className={cn('text-base font-bold', isToday && 'text-primary')}>{d.getDate()}</span>
   168→              <span className="h-1 w-1 rounded-full bg-primary/40" aria-hidden />
   169→            </button>
   170→          )
   171→        })}
   172→      </div>
   173→
   174→      {mode === 'day' && (
   175→        <Card>
   176→          <CardHeader className="pb-2">
   177→            <CardTitle className="flex items-center gap-2 text-sm">
   178→              <CalendarDays className="h-4 w-4 text-primary" />
   179→              Tasks for {new Date(`${selectedDay}T12:00:00`).toLocaleDateString('en-US', { weekday: 'long', month: 'short', day: 'numeric' })}
   180→            </CardTitle>
   181→          </CardHeader>
   182→          <CardContent className="space-y-1.5">
   183→            {dayTasks.loading ? (
   184→              <div className="space-y-2">{[1, 2].map((i) => <SkeletonCard key={i} className="h-10" />)}</div>
   185→            ) : (dayTasks.data?.tasks.length ?? 0) === 0 ? (
   186→              <p className="py-4 text-center text-sm text-muted-foreground">No tasks this day. Drag a task here from below, or add one.</p>
   187→            ) : (
   188→              dayTasks.data!.tasks.map((t) => (
   189→                <TaskRow key={t.id} task={t} onToggle={toggleTask} onSnooze={snoozeTask} />
   190→              ))
   191→            )}
   192→          </CardContent>
   193→        </Card>
   194→      )}
   195→
   196→      {/* ── Outline mode ── */}
   197→      {mode === 'outline' || mode === 'day' ? (
   198→        <div className="space-y-2">
   199→          {data.plans.length === 0 ? (
   200→            <EmptyState
   201→              icon={<CalendarDays className="h-5 w-5" />}
   202→              title="Build your first plan"
   203→              description="Create a yearly or monthly plan, nest weeks and days under it, then attach tasks. Or start from a template."
   204→              action={{ label: 'Browse templates', onClick: () => setTemplatesOpen(true) }}
   205→            />
   206→          ) : (
   207→            data.plans.map((p) => (
   208→              <PlanNodeRow
   209→                key={p.id}
   210→                node={p}
   211→                depth={0}
   212→                collapsed={collapsed}
   213→                toggleCollapse={toggleCollapse}
   214→                onToggleTask={toggleTask}
   215→                onSnoozeTask={snoozeTask}
   216→                onAddTask={addTaskToPlan}
   217→                onReload={reload}
   218→                allPlans={allPlansFlat}
   219→                mode={mode}
   220→              />
   221→            ))
   222→          )}
   223→        </div>
   224→      ) : null}
   225→
   226→      {/* ── Kanban mode ── */}
   227→      {mode === 'kanban' && <KanbanBoard plans={data.plans} onToggle={toggleTask} onReload={reload} />}
   228→
   229→      <AddPlanDialog open={addOpen} onOpenChange={setAddOpen} allPlans={allPlansFlat} onCreated={() => reload()} />
   230→      <TemplatesDialog open={templatesOpen} onOpenChange={setTemplatesOpen} onApplied={() => { reload(); toast({ title: 'Template applied — goal, plans & tasks created', description: 'Check Goals and Plans.' }) }} />
   231→    </div>
   232→  )
   233→}

   235→// ─── Plan title with one-click done toggle + inline rename ──────────
   236→function PlanTitleToggle({ node, onReload }: { node: PlanNode; onReload: () => void }) {
   237→  const [editing, setEditing] = useState(false)
   238→  const [title, setTitle] = useState(node.title)
   239→  const [busy, setBusy] = useState(false)
   240→  const { toast } = useToast()
   241→
   242→  // the row menu's "Rename" action focuses this row's editor
   243→  useEffect(() => {
   244→    const handler = (e: Event) => {
   245→      if ((e as CustomEvent).detail === node.id) setEditing(true)
   246→    }
   247→    document.addEventListener('cortex:rename-plan', handler)
   248→    return () => document.removeEventListener('cortex:rename-plan', handler)
   249→  }, [node.id])
   250→
   251→  async function toggleDone() {
   252→    try {
   253→      await api.patch(`/api/plans/${node.id}`, { done: !node.done })
   254→      onReload()
   255→    } catch {
   256→      toast({ title: 'Could not update plan', variant: 'destructive' })
   257→    }
   258→  }
   259→
   260→  async function save() {
   261→    const t = title.trim()
   262→    if (!t || t === node.title) return setEditing(false)
   263→    setBusy(true)
   264→    try {
   265→      await api.patch(`/api/plans/${node.id}`, { title: t })
   266→      setEditing(false)
   267→      onReload()
   268→    } catch {
   269→      toast({ title: 'Could not rename plan', variant: 'destructive' })
   270→    } finally {
   271→      setBusy(false)
   272→    }
   273→  }
   274→
   275→  if (editing) {
   276→    return (
   277→      <Input
   278→        value={title}
   279→        onChange={(e) => setTitle(e.target.value)}
   280→        onKeyDown={(e) => {
   281→          if (e.key === 'Enter') save()
   282→          if (e.key === 'Escape') {
   283→            setTitle(node.title)
   284→            setEditing(false)
   285→          }
   286→        }}
   287→        onBlur={save}
   288→        disabled={busy}
   289→        className="h-7 max-w-[240px] text-sm font-semibold"
   290→        autoFocus
   291→        aria-label="Plan title"
   292→      />
   293→    )
   294→  }
   295→
   296→  return (
   297→    <span className="flex min-w-0 items-center gap-1.5">
   298→      <button
   299→        onClick={toggleDone}
   300→        className={cn(
   301→          'flex h-5 w-5 shrink-0 items-center justify-center rounded-full border transition-colors',
   302→          node.done ? 'border-success bg-success text-white' : 'border-input text-transparent hover:border-success'
   303→        )}
   304→        aria-label={node.done ? `Mark "${node.title}" as not done` : `Mark "${node.title}" as done`}
   305→      >
   306→        <Check className="h-3 w-3" />
   307→      </button>
   308→      <span className={cn('truncate text-sm font-semibold', node.done && 'text-muted-foreground line-through')} title={node.title}>
   309→        {node.title}
   310→      </span>
   311→    </span>
   312→  )
   313→}

   344→// ─── Recursive plan node row ───────────────────────────────────────
   345→function PlanNodeRow({
   346→  node, depth, collapsed, toggleCollapse, onToggleTask, onSnoozeTask, onAddTask, onReload, allPlans, mode,
   347→}: {
   348→  node: PlanNode
   349→  depth: number
   350→  collapsed: Set<string>
   351→  toggleCollapse: (id: string) => void
   352→  onToggleTask: (t: Task) => void
   353→  onSnoozeTask: (t: Task) => void
   354→  onAddTask: (planId: string | null, title: string) => void
   355→  onReload: () => void
   356→  allPlans: { id: string; title: string; timeframe: string }[]
   357→  mode: string
   358→}) {
   359→  const [adding, setAdding] = useState(false)
   360→  const [newTask, setNewTask] = useState('')
   361→  const { toast } = useToast()
   362→  const hasChildren = (node.children?.length ?? 0) > 0
   363→  const isCollapsed = collapsed.has(node.id)
   364→  const tasks = node.tasks ?? []
   365→  const doneCount = tasks.filter((t) => t.status === 'done').length
   366→
   367→  const TF_ICON: Record<string, string> = { year: '🎯', month: '📅', week: '🗓', day: '☀️' }
   368→
   369→  return (
   370→    <div style={{ marginLeft: depth > 0 ? 16 : 0 }}>
   371→      <div className={cn('rounded-xl border bg-card transition-shadow hover:shadow-soft', depth === 0 && 'shadow-soft')}>
   372→        <div className="flex items-center gap-2 px-3 py-2.5">
   373→          {hasChildren ? (
   374→            <button onClick={() => toggleCollapse(node.id)} className="flex h-6 w-6 items-center justify-center rounded text-muted-foreground hover:bg-muted" aria-label={isCollapsed ? 'Expand' : 'Collapse'}>
   375→              {isCollapsed ? <ChevronRight className="h-4 w-4" /> : <ChevronDown className="h-4 w-4" />}
   376→            </button>
   377→          ) : (
   378→            <span className="w-6 text-center text-sm">{TF_ICON[node.timeframe]}</span>
   379→          )}
   380→          <PlanTitleToggle node={node} onReload={onReload} />
   381→          <Badgeish timeframe={node.timeframe} />
   382→          {node.goal && <span className="hidden rounded-full bg-sidebar-accent px-2 py-0.5 text-[10px] font-medium text-sidebar-accent-foreground sm:inline">{node.goal.title}</span>}
   383→          <span className="ml-auto text-[10px] tabular-nums text-muted-foreground">{doneCount}/{tasks.length} tasks</span>
   384→          <PlanRowMenu node={node} onReload={onReload} />
   385→          <button
   386→            onClick={async () => {
   387→              await api.del(`/api/plans/${node.id}`)
   388→              toast({ title: 'Plan deleted' })
   389→              onReload()
   390→            }}
   391→            className="flex h-7 w-7 items-center justify-center rounded-lg text-muted-foreground opacity-60 transition-opacity hover:text-danger"
   392→            aria-label="Delete plan"
   393→          >
   394→            <Trash2 className="h-3.5 w-3.5" />
   395→          </button>
   396→          <button onClick={() => setAdding((v) => !v)} className="flex h-7 w-7 items-center justify-center rounded-lg text-primary hover:bg-muted" aria-label="Add task to plan">
   397→            <Plus className="h-4 w-4" />
   398→          </button>
   399→        </div>
   400→
   401→        {adding && (
   402→          <div className="flex gap-2 px-3 pb-3">
   403→            <Input
   404→              value={newTask}
   405→              onChange={(e) => setNewTask(e.target.value)}
   406→              onKeyDown={(e) => {
   407→                if (e.key === 'Enter' && newTask.trim()) {
   408→                  onAddTask(node.id, newTask.trim())
   409→                  setNewTask('')
   410→                  setAdding(false)
   411→                }
   412→              }}
   413→              placeholder="Add a task to this plan…"
   414→              className="h-9"
   415→              autoFocus
   416→            />
   417→            <Button
   418→              size="sm"
   419→              className="h-9"
   420→              onClick={() => {
   421→                if (newTask.trim()) {
   422→                  onAddTask(node.id, newTask.trim())
   423→                  setNewTask('')
   424→                  setAdding(false)
   425→                }
   426→              }}
   427→            >
   428→              Add
   429→            </Button>
   430→          </div>
   431→        )}
   432→
   433→        <AnimatePresence initial={false}>
   434→          {!isCollapsed && (
   435→            <motion.div
   436→              initial={{ height: 0, opacity: 0 }}
   437→              animate={{ height: 'auto', opacity: 1 }}
   438→              exit={{ height: 0, opacity: 0 }}
   439→              transition={{ duration: 0.18, ease: 'easeOut' }}
   440→              className="overflow-hidden"
   441→            >
   442→              <div className="space-y-1 px-3 pb-3">
   443→                {tasks.map((t) => (
   444→                  <TaskRow key={t.id} task={t} onToggle={onToggleTask} onSnooze={onSnoozeTask} draggable />
   445→                ))}
   446→                {hasChildren && (node.children as PlanNode[]).map((child) => (
   447→                  <PlanNodeRow
   448→                    key={child.id}
   449→                    node={child}
   450→                    depth={depth + 1}
   451→                    collapsed={collapsed}
   452→                    toggleCollapse={toggleCollapse}
   453→                    onToggleTask={onToggleTask}
   454→                    onSnoozeTask={onSnoozeTask}
   455→                    onAddTask={onAddTask}
   456→                    onReload={onReload}
   457→                    allPlans={allPlans}
   458→                    mode={mode}
   459→                  />
   460→                ))}
   461→              </div>
   462→            </motion.div>
   463→          )}
   464→        </AnimatePresence>
   465→      </div>
   466→    </div>
   467→  )
   468→}

   470→function Badgeish({ timeframe }: { timeframe: string }) {

   475→  )
   476→}

   478→// ─── Task row with checkbox + drag handle ──────────────────────────
   479→function TaskRow({ task, onToggle, onSnooze, draggable }: { task: Task; onToggle: (t: Task) => void; onSnooze: (t: Task) => void; draggable?: boolean }) {
   480→  return (
   481→    <div
   482→      draggable={draggable}
   483→      onDragStart={(e) => {
   484→        e.dataTransfer.setData('text/task-id', task.id)
   485→        e.dataTransfer.effectAllowed = 'move'
   486→      }}
   487→      className={cn(
   488→        'group flex items-center gap-2.5 rounded-lg border bg-background px-3 py-2 transition-all hover:bg-muted/40',
   489→        task.status === 'done' && 'opacity-55'
   490→      )}
   491→    >
   492→      <Checkbox checked={task.status === 'done'} onCheckedChange={() => onToggle(task)} aria-label={`Toggle ${task.title}`} />
   493→      <PriorityDot priority={task.priority} />
   494→      <span className={cn('min-w-0 flex-1 truncate text-sm', task.status === 'done' && 'line-through')}>{task.title}</span>
   495→      {task.goal && <span className="hidden shrink-0 rounded-full bg-muted px-2 py-0.5 text-[10px] text-muted-foreground lg:inline">{task.goal.title}</span>}
   496→      {task.dueDate && (
   497→        <span className="hidden shrink-0 items-center gap-1 text-[10px] text-muted-foreground sm:flex">
   498→          <Clock className="h-3 w-3" />
   499→          {new Date(task.dueDate).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })}
   500→        </span>
   501→      )}
   502→      <button onClick={() => onSnooze(task)} className="flex h-6 w-6 shrink-0 items-center justify-center rounded text-muted-foreground opacity-0 transition-opacity hover:bg-muted group-hover:opacity-100" aria-label="Snooze to tomorrow">
   503→        <Clock className="h-3.5 w-3.5" />
   504→      </button>
   505→    </div>
   506→  )

   509→// ─── Kanban board (todo / doing / done) ────────────────────────────
   510→function KanbanBoard({ plans, onToggle, onReload }: { plans: PlanNode[]; onToggle: (t: Task) => void; onReload: () => void }) {
   511→  const columns: { key: string; label: string }[] = [
   512→    { key: 'todo', label: 'To do' },
   513→    { key: 'doing', label: 'In progress' },
   514→    { key: 'done', label: 'Done' },
   515→  ]
   516→  const allTasks = useMemo(() => {
   517→    const out: Task[] = []
   518→    const walk = (nodes: PlanNode[]) => {
   519→      for (const n of nodes) {
   520→        for (const t of n.tasks ?? []) out.push(t)
   521→        walk((n.children as PlanNode[]) ?? [])
   522→      }
   523→    }
   524→    walk(plans)
   525→    return out
   526→  }, [plans])
   527→
   528→  return (
   529→    <div className="grid gap-3 md:grid-cols-3">
   530→      {columns.map((col) => (
   531→        <div
   532→          key={col.key}
   533→          className="rounded-xl border bg-muted/30 p-2.5"
   534→          onDragOver={(e) => e.preventDefault()}
   535→          onDrop={async (e) => {
   536→            e.preventDefault()
   537→            const taskId = e.dataTransfer.getData('text/task-id')
   538→            if (taskId) {
   539→              await api.patch(`/api/tasks/${taskId}`, { status: col.key })
   540→              onReload()
   541→            }
   542→          }}
   543→        >
   544→          <p className="px-1 pb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   545→            {col.label} <span className="ml-1 text-[10px] font-normal">{allTasks.filter((t) => t.status === col.key).length}</span>
   546→          </p>
   547→          <div className="space-y-2">
   548→            {allTasks.filter((t) => t.status === col.key).map((t) => (
   549→              <motion.div key={t.id} layout initial={{ opacity: 0, scale: 0.97 }} animate={{ opacity: 1, scale: 1 }} transition={{ type: 'spring', stiffness: 380, damping: 30 }}>
   550→                <div
   551→                  draggable
   552→                  onDragStart={(e) => e.dataTransfer.setData('text/task-id', t.id)}
   553→                  className="cursor-grab rounded-lg border bg-card p-3 transition-shadow hover:shadow-soft active:cursor-grabbing"
   554→                >
   555→                  <div className="flex items-start gap-2">
   556→                    <Checkbox checked={t.status === 'done'} onCheckedChange={() => onToggle(t)} aria-label={`Toggle ${t.title}`} className="mt-0.5" />
   557→                    <div className="min-w-0 flex-1">
   558→                      <p className={cn('text-sm font-medium leading-snug', t.status === 'done' && 'line-through opacity-60')}>{t.title}</p>
   559→                      <div className="mt-1 flex items-center gap-2 text-[10px] text-muted-foreground">
   560→                        <PriorityDot priority={t.priority} /> ~{t.estimate}m
   561→                      </div>
   562→                    </div>
   563→                  </div>
   564→                </div>
   565→              </motion.div>
   566→            ))}
   567→            {allTasks.filter((t) => t.status === col.key).length === 0 && (
   568→              <p className="py-6 text-center text-xs text-muted-foreground">Drag tasks here</p>
   569→            )}
   570→          </div>
   571→        </div>
   572→      ))}
   573→    </div>
   574→  )
   575→}

   577→// ─── Add plan dialog ───────────────────────────────────────────────
   578→function AddPlanDialog({ open, onOpenChange, allPlans, onCreated }: { open: boolean; onOpenChange: (v: boolean) => void; allPlans: { id: string; title: string; timeframe: string }[]; onCreated: () => void }) {
   579→  const [title, setTitle] = useState('')
   580→  const [timeframe, setTimeframe] = useState('day')
   581→  const [parentId, setParentId] = useState('none')
   582→  const [startDate, setStartDate] = useState(todayISO())
   583→  const [busy, setBusy] = useState(false)
   584→  const { toast } = useToast()
   585→
   586→  async function create() {
   587→    if (!title.trim()) return
   588→    setBusy(true)
   589→    try {
   590→      await api.post('/api/plans', {
   591→        title: title.trim(),
   592→        timeframe,
   593→        parentId: parentId === 'none' ? null : parentId,
   594→        startDate: startDate || undefined,
   595→      })
   596→      setTitle('')
   597→      onOpenChange(false)
   598→      onCreated()
   599→    } catch {
   600→      toast({ title: 'Failed to create plan', variant: 'destructive' })
   601→    } finally {
   602→      setBusy(false)
   603→    }
   604→  }
   605→
   606→  return (
   607→    <Dialog open={open} onOpenChange={onOpenChange}>
   608→      <DialogContent className="sm:max-w-md">
   609→        <DialogHeader>
   610→          <DialogTitle>New plan</DialogTitle>
   611→          <DialogDescription>Nest it under a parent plan (year → month → week → day).</DialogDescription>
   612→        </DialogHeader>
   613→        <div className="space-y-3">
   614→          <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="e.g. September — interview sprint" aria-label="Plan title" autoFocus />
   615→          <div className="grid grid-cols-2 gap-3">
   616→            <div>
   617→              <label className="text-xs text-muted-foreground">Timeframe</label>
   618→              <Select value={timeframe} onValueChange={setTimeframe}>
   619→                <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   620→                <SelectContent>
   621→                  <SelectItem value="year">Year</SelectItem>
   622→                  <SelectItem value="month">Month</SelectItem>
   623→                  <SelectItem value="week">Week</SelectItem>
   624→                  <SelectItem value="day">Day</SelectItem>
   625→                </SelectContent>
   626→              </Select>
   627→            </div>
   628→            <div>
   629→              <label className="text-xs text-muted-foreground">Under parent</label>
   630→              <Select value={parentId} onValueChange={setParentId}>
   631→                <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   632→                <SelectContent>
   633→                  <SelectItem value="none">— none (top level) —</SelectItem>
   634→                  {allPlans.map((p) => (
   635→                    <SelectItem key={p.id} value={p.id}>{p.title}</SelectItem>
   636→                  ))}
   637→                </SelectContent>
   638→              </Select>
   639→            </div>
   640→          </div>
   641→          <div>
   642→            <label className="text-xs text-muted-foreground">Start date</label>
   643→            <Input type="date" value={startDate} onChange={(e) => setStartDate(e.target.value)} className="mt-1" />
   644→          </div>
   645→        </div>
   646→        <div className="flex justify-end gap-2">
   647→          <Button variant="ghost" onClick={() => onOpenChange(false)}>Cancel</Button>
   648→          <Button onClick={create} disabled={busy || !title.trim()}>
   649→            {busy && <Loader2 className="mr-1.5 h-4 w-4 animate-spin" />} Create plan
   650→          </Button>
   651→        </div>
   652→      </DialogContent>
   653→    </Dialog>
   654→  )
   655→}

   658→function TemplatesDialog({ open, onOpenChange, onApplied }: { open: boolean; onOpenChange: (v: boolean) => void; onApplied: () => void }) {

   670→      <DialogContent className="sm:max-w-lg">
   671→        <DialogHeader>

   675→        <div className="space-y-2">

   677→          {templates.map((t) => (
   678→            <button
   679→              key={t.id}
   680→              onClick={async () => {

   702→    </Dialog>
