   235→   235→// ─── Plan title with one-click done toggle + inline rename ──────────
   236→   236→function PlanTitleToggle({ node, onReload }: { node: PlanNode; onReload: () => void }) {
   237→   237→  const [editing, setEditing] = useState(false)
   238→   238→  const [title, setTitle] = useState(node.title)
   239→   239→  const [busy, setBusy] = useState(false)
   240→   240→  const { toast } = useToast()
   241→   241→
   242→   242→  // the row menu's "Rename" action focuses this row's editor
   243→   243→  useEffect(() => {
   244→   244→    const handler = (e: Event) => {
   245→   245→      if ((e as CustomEvent).detail === node.id) setEditing(true)
   246→   246→    }
   247→   247→    document.addEventListener('cortex:rename-plan', handler)
   248→   248→    return () => document.removeEventListener('cortex:rename-plan', handler)
   249→   249→  }, [node.id])
   250→   250→
   251→   251→  async function toggleDone() {
   252→   252→    try {
   253→   253→      await api.patch(`/api/plans/${node.id}`, { done: !node.done })
   254→   254→      onReload()
   255→   255→    } catch {
   256→   256→      toast({ title: 'Could not update plan', variant: 'destructive' })
   257→   257→    }
   258→   258→  }
   259→   259→
   260→   260→  async function save() {
   261→   261→    const t = title.trim()
   262→   262→    if (!t || t === node.title) return setEditing(false)
   263→   263→    setBusy(true)
   264→   264→    try {
   265→   265→      await api.patch(`/api/plans/${node.id}`, { title: t })
   266→   266→      setEditing(false)
   267→   267→      onReload()
   268→   268→    } catch {
   269→   269→      toast({ title: 'Could not rename plan', variant: 'destructive' })
   270→   270→    } finally {
   271→   271→      setBusy(false)
   272→   272→    }
   273→   273→  }
   274→   274→
   275→   275→  if (editing) {
   276→   276→    return (
   277→   277→      <Input
   278→   278→        value={title}
   279→   279→        onChange={(e) => setTitle(e.target.value)}
   280→   280→        onKeyDown={(e) => {
   281→   281→          if (e.key === 'Enter') save()
   282→   282→          if (e.key === 'Escape') {
   283→   283→            setTitle(node.title)
   284→   284→            setEditing(false)
   285→   285→          }
   286→   286→        }}
   287→   287→        onBlur={save}
   288→   288→        disabled={busy}
   289→   289→        className="h-7 max-w-[240px] text-sm font-semibold"
   290→   290→        autoFocus
   291→   291→        aria-label="Plan title"
   292→   292→      />
   293→   293→    )
   294→   294→  }
   295→   295→
   296→   296→  return (
   297→   297→    <span className="flex min-w-0 items-center gap-1.5">
   298→   298→      <button
   299→   299→        onClick={toggleDone}
   300→   300→        className={cn(
   301→   301→          'flex h-5 w-5 shrink-0 items-center justify-center rounded-full border transition-colors',
   302→   302→          node.done ? 'border-success bg-success text-white' : 'border-input text-transparent hover:border-success'
   303→   303→        )}
   304→   304→        aria-label={node.done ? `Mark "${node.title}" as not done` : `Mark "${node.title}" as done`}
   305→   305→      >
   306→   306→        <Check className="h-3 w-3" />
   307→   307→      </button>
   308→   308→      <span className={cn('truncate text-sm font-semibold', node.done && 'text-muted-foreground line-through')} title={node.title}>
   309→   309→        {node.title}
   310→   310→      </span>
   311→   311→    </span>
   312→   312→  )
   313→   313→}

   345→   345→function PlanNodeRow({
   346→   346→  node, depth, collapsed, toggleCollapse, onToggleTask, onSnoozeTask, onAddTask, onReload, allPlans, mode,
   347→   347→}: {
   348→   348→  node: PlanNode
   349→   349→  depth: number
   350→   350→  collapsed: Set<string>
   351→   351→  toggleCollapse: (id: string) => void
   352→   352→  onToggleTask: (t: Task) => void
   353→   353→  onSnoozeTask: (t: Task) => void
   354→   354→  onAddTask: (planId: string | null, title: string) => void
   355→   355→  onReload: () => void
   356→   356→  allPlans: { id: string; title: string; timeframe: string }[]
   357→   357→  mode: string
   358→   358→}) {
   359→   359→  const [adding, setAdding] = useState(false)
   360→   360→  const [newTask, setNewTask] = useState('')
   361→   361→  const { toast } = useToast()
   362→   362→  const hasChildren = (node.children?.length ?? 0) > 0
   363→   363→  const isCollapsed = collapsed.has(node.id)
   364→   364→  const tasks = node.tasks ?? []
   365→   365→  const doneCount = tasks.filter((t) => t.status === 'done').length
   366→   366→
   367→   367→  const TF_ICON: Record<string, string> = { year: '🎯', month: '📅', week: '🗓', day: '☀️' }
   368→   368→
   369→   369→  return (
   370→   370→    <div style={{ marginLeft: depth > 0 ? 16 : 0 }}>
   371→   371→      <div className={cn('rounded-xl border bg-card transition-shadow hover:shadow-soft', depth === 0 && 'shadow-soft')}>
   372→   372→        <div className="flex items-center gap-2 px-3 py-2.5">
   373→   373→          {hasChildren ? (
   374→   374→            <button onClick={() => toggleCollapse(node.id)} className="flex h-6 w-6 items-center justify-center rounded text-muted-foreground hover:bg-muted" aria-label={isCollapsed ? 'Expand' : 'Collapse'}>
   375→   375→              {isCollapsed ? <ChevronRight className="h-4 w-4" /> : <ChevronDown className="h-4 w-4" />}
   376→   376→            </button>
   377→   377→          ) : (
   378→   378→            <span className="w-6 text-center text-sm">{TF_ICON[node.timeframe]}</span>
   379→   379→          )}
   380→   380→          <PlanTitleToggle node={node} onReload={onReload} />
   381→   381→          <Badgeish timeframe={node.timeframe} />
   382→   382→          {node.goal && <span className="hidden rounded-full bg-sidebar-accent px-2 py-0.5 text-[10px] font-medium text-sidebar-accent-foreground sm:inline">{node.goal.title}</span>}
   383→   383→          <span className="ml-auto text-[10px] tabular-nums text-muted-foreground">{doneCount}/{tasks.length} tasks</span>
   384→   384→          <PlanRowMenu node={node} onReload={onReload} />
   385→   385→          <button
   386→   386→            onClick={async () => {
   387→   387→              await api.del(`/api/plans/${node.id}`)
   388→   388→              toast({ title: 'Plan deleted' })
   389→   389→              onReload()
   390→   390→            }}
   391→   391→            className="flex h-7 w-7 items-center justify-center rounded-lg text-muted-foreground opacity-60 transition-opacity hover:text-danger"
   392→   392→            aria-label="Delete plan"
   393→   393→          >
   394→   394→            <Trash2 className="h-3.5 w-3.5" />
   395→   395→          </button>
   396→   396→          <button onClick={() => setAdding((v) => !v)} className="flex h-7 w-7 items-center justify-center rounded-lg text-primary hover:bg-muted" aria-label="Add task to plan">
   397→   397→            <Plus className="h-4 w-4" />
   398→   398→          </button>
   399→   399→        </div>
   400→   400→
   401→   401→        {adding && (
   402→   402→          <div className="flex gap-2 px-3 pb-3">
   403→   403→            <Input
   404→   404→              value={newTask}
   405→   405→              onChange={(e) => setNewTask(e.target.value)}
   406→   406→              onKeyDown={(e) => {
   407→   407→                if (e.key === 'Enter' && newTask.trim()) {

   510→   510→function KanbanBoard({ plans, onToggle, onReload }: { plans: PlanNode[]; onToggle: (t: Task) => void; onReload: () => void }) {
   511→   511→  const columns: { key: string; label: string }[] = [
   512→   512→    { key: 'todo', label: 'To do' },
   513→   513→    { key: 'doing', label: 'In progress' },
   514→   514→    { key: 'done', label: 'Done' },
   515→   515→  ]
   516→   516→  const allTasks = useMemo(() => {
   517→   517→    const out: Task[] = []
   518→   518→    const walk = (nodes: PlanNode[]) => {
   519→   519→      for (const n of nodes) {
   520→   520→        for (const t of n.tasks ?? []) out.push(t)
   521→   521→        walk((n.children as PlanNode[]) ?? [])
   522→   522→      }
   523→   523→    }
   524→   524→    walk(plans)
   525→   525→    return out
   526→   526→  }, [plans])
   527→   527→
   528→   528→  return (
   529→   529→    <div className="grid gap-3 md:grid-cols-3">
   530→   530→      {columns.map((col) => (
   531→   531→        <div
   532→   532→          key={col.key}
   533→   533→          className="rounded-xl border bg-muted/30 p-2.5"
   534→   534→          onDragOver={(e) => e.preventDefault()}
   535→   535→          onDrop={async (e) => {
   536→   536→            e.preventDefault()
   537→   537→            const taskId = e.dataTransfer.getData('text/task-id')
   538→   538→            if (taskId) {
   539→   539→              await api.patch(`/api/tasks/${taskId}`, { status: col.key })
   540→   540→              onReload()
   541→   541→            }
   542→   542→          }}
   543→   543→        >
   544→   544→          <p className="px-1 pb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   545→   545→            {col.label} <span className="ml-1 text-[10px] font-normal">{allTasks.filter((t) => t.status === col.key).length}</span>
   546→   546→          </p>
   547→   547→          <div className="space-y-2">
   548→   548→            {allTasks.filter((t) => t.status === col.key).map((t) => (
   549→   549→              <motion.div key={t.id} layout initial={{ opacity: 0, scale: 0.97 }} animate={{ opacity: 1, scale: 1 }} transition={{ type: 'spring', stiffness: 380, damping: 30 }}>
   550→   550→                <div
   551→   551→                  draggable
   552→   552→                  onDragStart={(e) => e.dataTransfer.setData('text/task-id', t.id)}
   553→   553→                  className="cursor-grab rounded-lg border bg-card p-3 transition-shadow hover:shadow-soft active:cursor-grabbing"
   554→   554→                >
   555→   555→                  <div className="flex items-start gap-2">
   556→   556→                    <Checkbox checked={t.status === 'done'} onCheckedChange={() => onToggle(t)} aria-label={`Toggle ${t.title}`} className="mt-0.5" />
   557→   557→                    <div className="min-w-0 flex-1">
   558→   558→                      <p className={cn('text-sm font-medium leading-snug', t.status === 'done' && 'line-through opacity-60')}>{t.title}</p>
   559→   559→                      <div className="mt-1 flex items-center gap-2 text-[10px] text-muted-foreground">
   560→   560→                        <PriorityDot priority={t.priority} /> ~{t.estimate}m
   561→   561→                      </div>
   562→   562→                    </div>
   563→   563→                  </div>
   564→   564→                </div>
   565→   565→              </motion.div>
   566→   566→            ))}
   567→   567→            {allTasks.filter((t) => t.status === col.key).length === 0 && (
   568→   568→              <p className="py-6 text-center text-xs text-muted-foreground">Drag tasks here</p>
   569→   569→            )}
   570→   570→          </div>
   571→   571→        </div>
   572→   572→      ))}
   573→   573→    </div>
   574→   574→  )
   575→   575→}
