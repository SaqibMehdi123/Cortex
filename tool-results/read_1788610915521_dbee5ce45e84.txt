   196→   196→  function renderWithHighlights(text: string) {
   197→   197→    const highlights = doc!.highlights
   198→   198→    if (highlights.length === 0) return text
   199→   199→    for (const h of highlights) {
   200→   200→      const idx = text.indexOf(h.text.slice(0, Math.min(h.text.length, 80)))
   201→   201→      if (idx !== -1 && h.text.length <= text.length + 20) {
   202→   202→        const end = idx + h.text.length
   203→   203→        return (
   204→   204→          <>
   205→   205→            {text.slice(0, idx)}
   206→   206→            <mark className={cn('hl-' + h.color)} title={h.note ?? undefined}>
   207→   207→              {text.slice(idx, Math.min(end, text.length))}
   208→   208→            </mark>
   209→   209→            {text.slice(Math.min(end, text.length))}
   210→   210→          </>
   211→   211→        )
   212→   212→      }
   213→   213→    }
   214→   214→    return text
   215→   215→  }
