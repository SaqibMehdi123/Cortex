    32→// ─── Reader home (sidebar "Reader" tab) ─────────────────────────────
    33→// Distraction-free reading queue: pick up where you left off or start
    34→// anything from your reading list. Opens the full-screen Reader overlay.
    35→export function ReaderHome() {
    36→  const openReader = useUI((s) => s.openReader)
    37→  const setView = useUI((s) => s.setView)
    38→  const [docs, setDocs] = useState<DocumentItem[] | null>(null)
    39→
    40→  useEffect(() => {
    41→    api.get<{ documents: DocumentItem[] }>('/api/documents')
    42→      .then((d) => setDocs(d.documents))
    43→      .catch(() => setDocs([]))
    44→  }, [])
    45→
    46→  const reading = useMemo(
    47→    () => (docs ?? []).filter((d) => d.status === 'reading').sort((a, b) => (b.lastReadAt ?? b.updatedAt).localeCompare(a.lastReadAt ?? a.updatedAt)),
    48→    [docs]
    49→  )
    50→  const queued = useMemo(() => (docs ?? []).filter((d) => d.status === 'queued'), [docs])
    51→  const finishedCount = useMemo(() => (docs ?? []).filter((d) => d.status === 'finished').length, [docs])
    52→
    53→  if (!docs) {
    54→    return (
    55→      <div className="anim-fade-up space-y-4">
    56→        <div className="h-8 w-40 animate-pulse rounded bg-muted" />
    57→        <div className="grid gap-3 sm:grid-cols-2">{[1, 2, 3, 4].map((i) => <SkeletonCard key={i} />)}</div>
    58→      </div>
    59→    )
    60→  }
    61→
    62→  return (
    63→    <div className="anim-fade-up space-y-6 pb-8">
    64→      <PageHeader
    65→        icon={<BookOpen className="h-5 w-5" />}
    66→        title="Reader"
    67→        description="Your distraction-free reading queue — pick up anything and dive in."
    68→      />
    69→
    70→      {docs.length === 0 ? (
    71→        <EmptyState
    72→          icon={<BookOpen className="h-5 w-5" />}
    73→          title="Nothing to read yet"
    74→          description="The Reader is where you read what you save: import a PDF (with real text extraction), paste an article URL, or save a paper from News & Papers. Then read here with highlighting and Ask AI."
    75→          action={{ label: 'Import in Library', onClick: () => setView('library') }}
    76→        />
    77→      ) : (
    78→        <>
    79→          {reading.length > 0 && (
    80→            <section aria-label="Continue reading">
    81→              <h2 className="mb-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">Continue reading</h2>
    82→              <div className="grid gap-3 sm:grid-cols-2">
    83→                {reading.slice(0, 4).map((doc) => (
    84→                  <button key={doc.id} onClick={() => openReader(doc.id)} className="text-left">
    85→                    <Card className="card-lift h-full">
    86→                      <CardContent className="space-y-2 p-4">
    87→                        <div className="flex items-center gap-2 text-xs text-muted-foreground">
    88→                          <Badge variant="outline" className="text-[10px] capitalize">{doc.type}</Badge>
    89→                          {doc.author && <span className="truncate">{doc.author}</span>}
    90→                        </div>
    91→                        <p className="line-clamp-2 text-sm font-semibold leading-snug">{doc.title}</p>
    92→                        <div className="flex items-center gap-2 pt-1">
    93→                          <Progress value={doc.progress} className="h-1.5" />
    94→                          <span className="shrink-0 text-[10px] text-muted-foreground">{doc.progress}%</span>
    95→                        </div>
    96→                      </CardContent>
    97→                    </Card>
    98→                  </button>
    99→                ))}
   100→              </div>
   101→            </section>
   102→          )}
   103→
   104→          <section aria-label="Reading queue">
   105→            <div className="mb-2 flex items-center justify-between">
   106→              <h2 className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Reading queue</h2>
   107→              <span className="text-xs text-muted-foreground">{queued.length} waiting · {finishedCount} finished</span>
   108→            </div>
   109→            {queued.length === 0 ? (
   110→              <div className="rounded-xl border border-dashed px-4 py-8 text-center text-sm text-muted-foreground">
   111→                Queue is clear — everything saved is in progress or finished.
   112→              </div>
   113→            ) : (
   114→              <div className="overflow-hidden rounded-xl border bg-card">
   115→                {queued.map((doc, i) => (
   116→                  <button
   117→                    key={doc.id}
   118→                    onClick={() => openReader(doc.id)}
   119→                    className={cn('flex w-full items-center gap-3 px-4 py-3 text-left transition-colors hover:bg-muted/50', i !== 0 && 'border-t')}
   120→                  >
   121→                    <BookOpen className="h-4 w-4 shrink-0 text-primary" />
   122→                    <span className="min-w-0 flex-1">
   123→                      <span className="block truncate text-sm font-medium">{doc.title}</span>
   124→                      {doc.source && <span className="block truncate text-xs text-muted-foreground">{doc.source}</span>}
   125→                    </span>
   126→                    <ChevronRight className="h-4 w-4 shrink-0 text-muted-foreground" />
   127→                  </button>
   128→                ))}
   129→              </div>
   130→            )}
   131→          </section>
   132→        </>
   133→      )}
   134→    </div>
   135→  )
   136→}

   138→export function Reader() {
   139→  const readerDocId = useUI((s) => s.readerDocId)
   140→  const closeReader = useUI((s) => s.closeReader)
   141→  const { toast } = useToast()
   142→
   143→  const [doc, setDoc] = useState<(DocumentItem & { highlights: Highlight[] }) | null>(null)
   144→  const [loading, setLoading] = useState(true)
   145→  const [selection, setSelection] = useState<SelInfo | null>(null)
   146→  const [aiOpen, setAiOpen] = useState(false)
   147→  const [messages, setMessages] = useState<ChatMessage[]>([])
   148→  const [aiInput, setAiInput] = useState('')
   149→  const [aiBusy, setAiBusy] = useState(false)
   150→  const [summarizing, setSummarizing] = useState(false)
   151→  const [noteDialogFor, setNoteDialogFor] = useState<string | null>(null)
   152→  const [noteText, setNoteText] = useState('')
   153→  const [mindmapOpen, setMindmapOpen] = useState(false)
   154→  const [mindmaps, setMindmaps] = useState<{ id: string; title: string }[]>([])
   155→  const [mindmapChoice, setMindmapChoice] = useState<string>('__new')
   156→  const [lastHighlightId, setLastHighlightId] = useState<string | null>(null)
   157→  const [flashcardBusy, setFlashcardBusy] = useState(false)
   158→
   159→  const contentRef = useRef<HTMLDivElement>(null)
   160→  const scrollRef = useRef<HTMLDivElement>(null)
   161→  const sessionStart = useRef<number>(Date.now())
   162→  const lastLogged = useRef<number>(Date.now())
   163→
   164→  // Load document + chat
   165→  useEffect(() => {
   166→    if (!readerDocId) return
   167→    setLoading(true)
   168→    sessionStart.current = Date.now()
   169→    lastLogged.current = Date.now()
   170→    Promise.all([
   171→      api.get<{ document: DocumentItem & { highlights: Highlight[] } }>(`/api/documents/${readerDocId}`),
   172→      api.get<{ messages: ChatMessage[] }>(`/api/chat?documentId=${readerDocId}`),
   173→      api.get<{ mindmaps: { id: string; title: string }[] }>('/api/mindmaps'),
   174→    ])
   175→      .then(([d, m, mm]) => {
   176→        setDoc(d.document)
   177→        setMessages(m.messages)
   178→        setMindmaps(mm.mindmaps)
   179→      })
   180→      .catch(() => toast({ title: 'Failed to open document', variant: 'destructive' }))
   181→      .finally(() => setLoading(false))
   182→  }, [readerDocId, toast])
   183→
   184→  // Log reading time every minute + update lastReadAt
   185→  useEffect(() => {
   186→    if (!readerDocId || !doc) return
   187→    const interval = setInterval(async () => {
   188→      const minutes = Math.round((Date.now() - lastLogged.current) / 60000)
   189→      if (minutes >= 1) {
   190→        lastLogged.current = Date.now()
   191→        try {
   192→          await api.post('/api/reading-sessions', { documentId: readerDocId, minutes })
   193→        } catch {}
   194→      }
   195→    }, 60000)
   196→    return () => {
   197→      clearInterval(interval)
   198→      // flush remainder on unmount
   199→      const minutes = Math.round((Date.now() - lastLogged.current) / 60000)
   200→      if (minutes >= 1) {
   201→        api.post('/api/reading-sessions', { documentId: readerDocId, minutes }).catch(() => {})
   202→      }
   203→    }
   204→  }, [readerDocId, doc])
   205→
   206→  // Scroll-driven progress
   207→  const onScroll = useCallback(() => {
   208→    const el = scrollRef.current
   209→    if (!el || !doc) return
   210→    const pct = Math.min(100, Math.round((el.scrollTop / Math.max(1, el.scrollHeight - el.clientHeight)) * 100))
   211→    if (pct > doc.progress) {
   212→      setDoc({ ...doc, progress: pct })
   213→      api.patch(`/api/documents/${doc.id}`, { progress: pct, status: doc.status === 'queued' ? 'reading' : doc.status }).catch(() => {})
   214→    }
   215→  }, [doc])
   216→
   217→  // Selection handling
   218→  useEffect(() => {
   219→    const handler = () => {
   220→      const sel = window.getSelection()
   221→      if (!sel || sel.isCollapsed || !contentRef.current) return setSelection(null)
   222→      const text = sel.toString().trim()
   223→      if (text.length < 2) return setSelection(null)
   224→      const range = sel.getRangeAt(0)
   225→      const rect = range.getBoundingClientRect()
   226→      setSelection({ text, x: rect.left + rect.width / 2, y: rect.top })
   227→    }
   228→    document.addEventListener('mouseup', handler)
   229→    document.addEventListener('touchend', handler)
   230→    return () => {
   231→      document.removeEventListener('mouseup', handler)
   232→      document.removeEventListener('touchend', handler)
   233→    }
   234→  }, [])
   235→
   236→  async function addHighlight(color: (typeof HL_COLORS)[number]) {
   237→    if (!selection || !doc) return
   238→    const text = selection.text
   239→    const position = doc.content?.indexOf(text.slice(0, 60)) ?? 0
   240→    setSelection(null)
   241→    window.getSelection()?.removeAllRanges()
   242→    try {
   243→      const { highlight } = await api.post<{ highlight: Highlight }>('/api/highlights', {
   244→        documentId: doc.id, text, color, position: Math.max(0, position),
   245→      })
   246→      setDoc({ ...doc, highlights: [...doc.highlights, highlight] })
   247→      setLastHighlightId(highlight.id)
   248→      toast({ title: 'Highlighted', description: 'Turn it into a flashcard or mindmap node anytime.' })
   249→    } catch {
   250→      toast({ title: 'Could not save highlight', variant: 'destructive' })
   251→    }
   252→  }
   253→
   254→  function renderContent() {
   255→    if (!doc?.content) {
   256→      return (
   257→        <div className="rounded-xl border border-dashed p-8 text-center">
   258→          <BookOpen className="mx-auto h-8 w-8 text-muted-foreground/50" />
   259→          <p className="mt-3 font-medium">No text content yet</p>
   260→          <p className="mx-auto mt-1 max-w-sm text-sm text-muted-foreground">
   261→            This entry is metadata-only (e.g. a PDF or book). Paste its text below to unlock highlighting and AI Q&amp;A.
   262→          </p>
   263→          <Textarea
   264→            className="mx-auto mt-4 min-h-[160px] max-w-xl text-left"
   265→            placeholder="Paste the full text…"
   266→            aria-label="Paste document text"
   267→            onBlur={(e) => {
   268→              const v = e.target.value.trim()
   269→              if (v && doc) {
   270→                api.patch(`/api/documents/${doc.id}`, { content: v }).then(() => setDoc({ ...doc, content: v }))
   271→              }
   272→            }}
   273→          />
   274→        </div>
   275→      )
   276→    }
   277→
   278→    // Render paragraphs, wrapping highlighted ranges via simple text matching per paragraph
   279→    const paragraphs = doc.content.split(/\n{1,}/)
   280→    return paragraphs.map((p, i) => <p key={i}>{renderWithHighlights(p)}</p>)
   281→  }
   282→
   283→  function renderWithHighlights(text: string) {
   284→    const highlights = doc!.highlights
   285→    if (highlights.length === 0) return text
   286→    // find highlight that is contained in this text (first match wins)
   287→    for (const h of highlights) {
   288→      const idx = text.indexOf(h.text.slice(0, Math.min(h.text.length, 80)))
   289→      if (idx !== -1 && h.text.length <= text.length + 20) {
   290→        const end = idx + h.text.length
   291→        return (
   292→          <>
   293→            {text.slice(0, idx)}
   294→            <mark className={cn('hl-' + h.color)} title={h.note ?? undefined}>
   295→              {text.slice(idx, Math.min(end, text.length))}
   296→            </mark>
   297→            {text.slice(Math.min(end, text.length))}
   298→          </>
   299→        )
   300→      }
   301→    }
   302→    return text
   303→  }
   304→
   305→  async function summarize() {
   306→    if (!doc) return
   307→    setSummarizing(true)
   308→    try {
   309→      const { document: updated } = await api.post<{ document: DocumentItem }>(`/api/documents/${doc.id}/summarize`)
   310→      setDoc({ ...doc, ...updated, highlights: doc.highlights })
   311→      toast({ title: 'Summary ready' })
   312→    } catch (e) {
   313→      toast({ title: 'Summary failed', description: e instanceof Error ? e.message : 'No text content?', variant: 'destructive' })
   314→    } finally {
   315→      setSummarizing(false)
   316→    }
   317→  }
   318→
   319→  async function sendAI(text?: string) {
   320→    const msg = (text ?? aiInput).trim()
   321→    if (!msg || !doc || aiBusy) return
   322→    setAiInput('')
   323→    setAiBusy(true)
   324→    setMessages((prev) => [...prev, { id: `tmp-${Date.now()}`, documentId: doc.id, role: 'user', content: msg, citations: null, createdAt: new Date().toISOString() }])
   325→    try {
   326→      const d = await api.post<{ assistantMessage: ChatMessage }>('/api/chat', { documentId: doc.id, message: msg })
   327→      setMessages((prev) => [...prev, d.assistantMessage])
   328→    } catch {
   329→      toast({ title: 'AI request failed', variant: 'destructive' })
   330→    } finally {
   331→      setAiBusy(false)
   332→    }
   333→  }
   334→
   335→  async function makeFlashcard() {
   336→    if (!doc || !lastHighlightId) return
   337→    setFlashcardBusy(true)
   338→    try {
   339→      await api.post('/api/flashcards', { generate: true, highlightId: lastHighlightId })
   340→      toast({ title: 'Flashcard created', description: 'Find it in Flashcards for spaced review.' })
   341→    } catch {
   342→      toast({ title: 'Flashcard generation failed', variant: 'destructive' })
   343→    } finally {
   344→      setFlashcardBusy(false)
   345→    }
   346→  }
   347→
   348→  async function addToMindmap() {
   349→    if (!doc) return
   350→    try {
   351→      if (mindmapChoice === '__new') {
   352→        await api.post('/api/mindmaps/generate', { documentId: doc.id })
   353→        toast({ title: 'Mindmap generated', description: 'Open Mindmaps to explore it.' })
   354→      } else {
   355→        // add node to chosen map
   356→        const maps = await api.get<{ mindmaps: { id: string; nodes: { x: number; y: number }[] }[] }>('/api/mindmaps')
   357→        const target = maps.mindmaps.find((m) => m.id === mindmapChoice)
   358→        const maxY = target?.nodes.reduce((mx, n) => Math.max(mx, n.y), 0) ?? 0
   359→        const nodes = target?.nodes ?? []
   360→        await api.patch(`/api/mindmaps/${mindmapChoice}`, {
   361→          nodes: [
   362→            ...nodes,
   363→            { id: `n-${Date.now()}`, label: doc.title.slice(0, 40), x: 120, y: maxY + 90, parentId: null, color: 'teal', linkType: 'document', linkId: doc.id },
   364→          ],
   365→        })
   366→        toast({ title: 'Added to mindmap' })
   367→      }
   368→      setMindmapOpen(false)
   369→    } catch {
   370→      toast({ title: 'Mindmap update failed', variant: 'destructive' })
   371→    }
   372→  }
   373→
   374→  const takeaways = useMemo<string[]>(() => {
   375→    if (!doc?.takeaways) return []
   376→    try { return JSON.parse(doc.takeaways) } catch { return [] }
   377→  }, [doc?.takeaways])
   378→
   379→  if (!readerDocId) return null
   380→
   381→  return (
   382→    <div className="fixed inset-0 z-40 flex bg-background" role="dialog" aria-label="Reader">
   383→      {/* ── Main reading column ── */}
   384→      <div className="flex min-w-0 flex-1 flex-col">
   385→        {/* Header */}
   386→        <header className="flex items-center gap-2 border-b px-3 py-2.5 sm:px-5">
   387→          <Button variant="ghost" size="icon" className="h-9 w-9" onClick={closeReader} aria-label="Back to library">
   388→            <ArrowLeft className="h-4.5 w-4.5" />
   389→          </Button>
   390→          <div className="min-w-0 flex-1">
   391→            <p className="truncate text-sm font-semibold">{doc?.title ?? 'Loading…'}</p>
   392→            {doc?.author && <p className="truncate text-xs text-muted-foreground">{doc.author}</p>}
   393→          </div>
   394→          {doc && (
   395→            <>
   396→              <Button variant="outline" size="sm" className="hidden h-8 text-xs sm:flex" onClick={summarize} disabled={summarizing}>
   397→                {summarizing ? <Loader2 className="mr-1.5 h-3.5 w-3.5 animate-spin" /> : <Sparkles className="mr-1.5 h-3.5 w-3.5" />}
   398→                Summarize
   399→              </Button>
   400→              <Button
   401→                variant="outline"
   402→                size="sm"
   403→                className="h-8 text-xs"
   404→                onClick={async () => {
   405→                  const finished = doc.status === 'finished'
   406→                  const { document: updated } = await api.patch<{ document: DocumentItem }>(`/api/documents/${doc.id}`, {
   407→                    status: finished ? 'reading' : 'finished',
   408→                    progress: finished ? doc.progress : 100,
   409→                  })
   410→                  setDoc({ ...doc, ...updated, highlights: doc.highlights })
   411→                  if (!finished) toast({ title: 'Marked as finished 🎉' })
   412→                }}
   413→              >
   414→                <CheckCircle2 className={cn('mr-1.5 h-3.5 w-3.5', doc.status === 'finished' && 'text-success')} />
   415→                {doc.status === 'finished' ? 'Reopen' : 'Finish'}
   416→              </Button>
   417→              <Button size="sm" className="h-8 text-xs" onClick={() => setAiOpen((v) => !v)}>
   418→                <Sparkles className="mr-1.5 h-3.5 w-3.5" /> Ask AI
   419→              </Button>
   420→            </>
   421→          )}
   422→        </header>
   423→        {doc && (
   424→          <div className="flex items-center gap-3 border-b px-4 py-1.5 sm:px-6">
   425→            <Progress value={doc.progress} className="h-1" />
   426→            <span className="shrink-0 text-[10px] text-muted-foreground">{doc.progress}%</span>
   427→          </div>
   428→        )}
   429→
   430→        {/* Content */}
   431→        <div ref={scrollRef} onScroll={onScroll} className="scroll-thin relative flex-1 overflow-y-auto" style={{ transition: 'scroll 150ms ease' }}>
   432→          <div className={cn('mx-auto px-5 py-8 sm:px-8', aiOpen ? 'max-w-none' : 'max-w-[680px]')}>
   433→            {loading ? (
   434→              <div className="space-y-4 py-8">
   435→                {Array.from({ length: 8 }).map((_, i) => (
   436→                  <div key={i} className="h-4 animate-pulse rounded bg-muted" style={{ width: `${70 + ((i * 13) % 30)}%` }} />
   437→                ))}
   438→              </div>
   439→            ) : (
   440→              <>
   441→                {/* Summary block */}
   442→                {doc?.summary && (
   443→                  <div className="anim-pop mb-8 rounded-xl border bg-card p-4">
   444→                    <div className="flex items-center gap-2 text-xs font-semibold text-primary">
   445→                      <Sparkles className="h-3.5 w-3.5" /> AI summary
   446→                    </div>
   447→                    <p className="mt-2 text-sm leading-relaxed text-muted-foreground">{doc.summary}</p>
   448→                    {takeaways.length > 0 && (
   449→                      <ul className="mt-3 space-y-1.5">
   450→                        {takeaways.map((t, i) => (
   451→                          <li key={i} className="flex gap-2 text-sm"><span className="text-success">✓</span><span>{t}</span></li>
   452→                        ))}
   453→                      </ul>
   454→                    )}
   455→                  </div>
   456→                )}
   457→
   458→                <div ref={contentRef} className="reading-prose">
   459→                  {doc && renderContent()}
   460→                </div>
   461→                <p className="mt-12 text-center text-xs text-muted-foreground">— end —</p>
   462→              </>
   463→            )}
   464→          </div>
   465→        </div>
   466→      </div>
   467→
   468→      {/* ── Ask AI side panel ── */}
   469→      {aiOpen && doc && (
   470→        <aside className="fixed inset-y-0 right-0 z-50 flex w-full flex-col border-l bg-background sm:w-[400px]" aria-label="Document AI chat">
   471→          <div className="flex items-center gap-2 border-b px-4 py-3">
   472→            <Sparkles className="h-4 w-4 text-primary" />
   473→            <p className="flex-1 text-sm font-semibold">Ask AI about this document</p>
   474→            <Button
   475→              variant="ghost"
   476→              size="icon"
   477→              className="h-8 w-8"
   478→              aria-label="Clear chat"
   479→              onClick={async () => {
   480→                await api.del(`/api/chat?documentId=${doc.id}`)
   481→                setMessages([])
   482→              }}
   483→            >
   484→              <Trash2 className="h-4 w-4" />
   485→            </Button>
   486→            <Button variant="ghost" size="icon" className="h-8 w-8" aria-label="Close chat" onClick={() => setAiOpen(false)}>
   487→              <X className="h-4 w-4" />
   488→            </Button>
   489→          </div>
   490→
   491→          <ScrollArea className="scroll-thin flex-1 px-4 py-4">
   492→            <div className="space-y-3">
   493→              {messages.length === 0 && (
   494→                <div className="space-y-2">
   495→                  <p className="text-sm text-muted-foreground">Ask anything — answers cite the document.</p>
   496→                  {['Summarize the key argument', 'What methodology was used?', 'Explain the hardest concept simply'].map((s) => (
   497→                    <button key={s} onClick={() => sendAI(s)} className="block w-full rounded-lg border bg-card px-3 py-2 text-left text-xs transition-colors hover:bg-muted">
   498→                      {s}
   499→                    </button>
   500→                  ))}
   501→                </div>
   502→              )}
   503→              {messages.map((m) => (
   504→                <div key={m.id} className={cn('flex', m.role === 'user' ? 'justify-end' : 'justify-start')}>
   505→                  <div className={cn('max-w-[88%] rounded-2xl px-3 py-2 text-sm', m.role === 'user' ? 'rounded-br-md bg-primary text-primary-foreground' : 'rounded-bl-md border bg-card')}>
   506→                    {m.role === 'user' ? (
   507→                      m.content
   508→                    ) : (
   509→                      <>
   510→                        <div className="[&_code]:rounded [&_code]:bg-muted [&_code]:px-1 [&_li]:ml-4 [&_li]:list-disc [&_p]:mb-2 [&_strong]:font-semibold [&_ul]:ml-4 [&_ul]:list-disc">
   511→                          <ReactMarkdown>{m.content}</ReactMarkdown>
   512→                        </div>
   513→                        {m.citations?.length ? (
   514→                          <div className="mt-2 space-y-1 border-t pt-2">
   515→                            {m.citations.map((c: Citation) => (
   516→                              <p key={c.n} className="text-[10px] leading-snug text-muted-foreground">
   517→                                <span className="mr-1 inline-flex h-4 w-4 items-center justify-center rounded-full bg-primary/15 text-[9px] font-bold text-primary">{c.n}</span>
   518→                                {c.label}
   519→                              </p>
   520→                            ))}
   521→                          </div>
   522→                        ) : null}
   523→                      </>
   524→                    )}
   525→                  </div>
   526→                </div>
   527→              ))}
   528→              {aiBusy && (
   529→                <div className="flex items-center gap-2 text-xs text-muted-foreground">
   530→                  <Loader2 className="h-3.5 w-3.5 animate-spin" /> Reading the document…
   531→                </div>
   532→              )}
   533→            </div>
   534→          </ScrollArea>
   535→
   536→          <div className="border-t p-3">
   537→            <div className="flex items-end gap-2">
   538→              <Textarea
   539→                value={aiInput}
   540→                onChange={(e) => setAiInput(e.target.value)}
   541→                onKeyDown={(e) => {
   542→                  if (e.key === 'Enter' && !e.shiftKey) {
   543→                    e.preventDefault()
   544→                    sendAI()
   545→                  }
   546→                }}
   547→                placeholder="Ask about this document…"
   548→                className="min-h-[44px] flex-1 resize-none"
   549→                rows={1}
   550→                aria-label="Ask AI"
   551→              />
   552→              <Button size="icon" className="h-11 w-11" onClick={() => sendAI()} disabled={aiBusy || !aiInput.trim()} aria-label="Send question">
   553→                <Send className="h-4 w-4" />
   554→              </Button>
   555→            </div>
   556→          </div>
   557→        </aside>
   558→      )}
   559→
   560→      {/* ── Floating selection toolbar ── */}
   561→      {selection && (
   562→        <div
   563→          className="anim-pop fixed z-50 flex -translate-x-1/2 -translate-y-full items-center gap-1 rounded-xl border bg-popover p-1.5 shadow-lg"
   564→          style={{ left: selection.x, top: selection.y - 8 }}
   565→          role="toolbar"
   566→          aria-label="Highlight actions"
   567→        >
   568→          {HL_COLORS.map((c) => (
   569→            <button
   570→              key={c}
   571→              onClick={() => addHighlight(c)}
   572→              className={cn('h-7 w-7 rounded-full border-2 transition-transform hover:scale-110', {
   573→                'bg-yellow-400/80': c === 'yellow',
   574→                'bg-green-500/70': c === 'green',
   575→                'bg-blue-500/60': c === 'blue',
   576→                'bg-pink-500/60': c === 'pink',
   577→              })}
   578→              aria-label={`Highlight ${c}`}
   579→            />
   580→          ))}
   581→          <span className="mx-1 h-5 w-px bg-border" />
   582→          <button
   583→            onClick={() => {
   584→              setNoteDialogFor('new')
   585→              setNoteText('')
   586→            }}
   587→            className="flex h-7 w-7 items-center justify-center rounded-lg text-muted-foreground transition-colors hover:bg-muted"
   588→            aria-label="Add note to highlight"
   589→          >
   590→            <StickyNote className="h-4 w-4" />
   591→          </button>
   592→          <button
   593→            onClick={() => {
   594→              addHighlight('yellow').then(() => setMindmapOpen(true))
   595→            }}
   596→            className="flex h-7 w-7 items-center justify-center rounded-lg text-muted-foreground transition-colors hover:bg-muted"
   597→            aria-label="Add to mindmap"
   598→          >
   599→            <Share2 className="h-4 w-4" />
   600→          </button>
   601→          <button
   602→            onClick={() => {
   603→              addHighlight('yellow').then(() => makeFlashcard())
   604→            }}
   605→            className="flex h-7 items-center gap-1 rounded-lg px-2 text-xs font-medium text-primary transition-colors hover:bg-muted"
   606→            disabled={flashcardBusy}
   607→          >
   608→            {flashcardBusy ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Layers className="h-3.5 w-3.5" />} Flashcard
   609→          </button>
   610→        </div>
   611→      )}
   612→
   613→      {/* Note dialog */}
   614→      <Dialog open={noteDialogFor !== null} onOpenChange={(v) => !v && setNoteDialogFor(null)}>
   615→        <DialogContent className="sm:max-w-md">
   616→          <DialogHeader>
   617→            <DialogTitle>Attach a note</DialogTitle>
   618→            <DialogDescription className="line-clamp-2">{selection?.text ?? doc?.highlights.find((h) => h.id === lastHighlightId)?.text}</DialogDescription>
   619→          </DialogHeader>
   620→          <Textarea value={noteText} onChange={(e) => setNoteText(e.target.value)} placeholder="Your thought on this passage…" className="min-h-[100px]" autoFocus />
   621→          <div className="flex justify-end gap-2">
   622→            <Button variant="ghost" onClick={() => setNoteDialogFor(null)}>Cancel</Button>
   623→            <Button
   624→              onClick={async () => {
   625→                if (!doc) return
   626→                try {
   627→                  // save selection as highlight with note
   628→                  const text = selection?.text ?? ''
   629→                  if (text) {
   630→                    const position = Math.max(0, doc.content?.indexOf(text.slice(0, 60)) ?? 0)
   631→                    const { highlight } = await api.post<{ highlight: Highlight }>('/api/highlights', { documentId: doc.id, text, color: 'blue', note: noteText, position })
   632→                    setDoc({ ...doc, highlights: [...doc.highlights, highlight] })
   633→                    setLastHighlightId(highlight.id)
   634→                  } else if (lastHighlightId) {
   635→                    await api.patch(`/api/highlights/${lastHighlightId}`, { note: noteText })
   636→                    setDoc({
   637→                      ...doc,
   638→                      highlights: doc.highlights.map((h) => (h.id === lastHighlightId ? { ...h, note: noteText } : h)),
   639→                    })
   640→                  }
   641→                  setNoteDialogFor(null)
   642→                  setSelection(null)
   643→                  window.getSelection()?.removeAllRanges()
   644→                  toast({ title: 'Note attached' })
   645→                } catch {
   646→                  toast({ title: 'Failed to save note', variant: 'destructive' })
   647→                }
   648→              }}
   649→            >
   650→              Save note
   651→            </Button>
   652→          </div>
   653→        </DialogContent>
   654→      </Dialog>
   655→
   656→      {/* Mindmap dialog */}
   657→      <Dialog open={mindmapOpen} onOpenChange={setMindmapOpen}>
   658→        <DialogContent className="sm:max-w-sm">
   659→          <DialogHeader>
   660→            <DialogTitle>Add to mindmap</DialogTitle>
   661→            <DialogDescription>Append this document to an existing map, or auto-generate a fresh one from its content.</DialogDescription>
   662→          </DialogHeader>
   663→          <Select value={mindmapChoice} onValueChange={setMindmapChoice}>
   664→            <SelectTrigger aria-label="Choose mindmap">
   665→              <SelectValue />
   666→            </SelectTrigger>
   667→            <SelectContent>
   668→              <SelectItem value="__new">✨ Generate new mindmap from this document</SelectItem>
   669→              {mindmaps.map((m) => (
   670→                <SelectItem key={m.id} value={m.id}>{m.title}</SelectItem>
   671→              ))}
   672→            </SelectContent>
   673→          </Select>
   674→          <div className="flex justify-end gap-2">
   675→            <Button variant="ghost" onClick={() => setMindmapOpen(false)}>Cancel</Button>
   676→            <Button onClick={addToMindmap}>
   677→              <Share2 className="mr-1.5 h-4 w-4" /> Add
   678→            </Button>
   679→          </div>
   680→        </DialogContent>
   681→      </Dialog>
   682→    </div>
   683→  )
   684→}
