   229→   229→    document.addEventListener('touchend', handler)
   230→   230→    return () => {
   231→   231→      document.removeEventListener('mouseup', handler)
   232→   232→      document.removeEventListener('touchend', handler)
   233→   233→    }

   283→   283→  function renderWithHighlights(text: string) {
   284→   284→    const highlights = doc!.highlights
   285→   285→    if (highlights.length === 0) return text
   286→   286→    // find highlight that is contained in this text (first match wins)
   287→   287→    for (const h of highlights) {
   288→   288→      const idx = text.indexOf(h.text.slice(0, Math.min(h.text.length, 80)))
   289→   289→      if (idx !== -1 && h.text.length <= text.length + 20) {
   290→   290→        const end = idx + h.text.length
   291→   291→        return (
   292→   292→          <>
   293→   293→            {text.slice(0, idx)}
   294→   294→            <mark className={cn('hl-' + h.color)} title={h.note ?? undefined}>
   295→   295→              {text.slice(idx, Math.min(end, text.length))}
   296→   296→            </mark>
   297→   297→            {text.slice(Math.min(end, text.length))}
   298→   298→          </>
   299→   299→        )
   300→   300→      }
   301→   301→    }
   302→   302→    return text
   303→   303→  }
